"""CAD-PART HIERARCHY V0.7.8 - V0.7.2 CSV -> hierarchy Excel (offline).
Does not launch AutoCAD or change cadpart_v03.py and its extraction output.
Run: python cadpart_hierarchy_v01.py
Python standard library only (no pip / openpyxl required).
"""
from pathlib import Path
from collections import defaultdict, Counter
from xml.sax.saxutils import escape
from datetime import datetime
import csv
import json
import re
import sys
import zipfile

VERSION = "0.7.8"
ROOT = Path(__file__).resolve().parent
RUNS = ROOT / "CAD_RESULT" / "RUNS"
REQUIRED = ("DRAWING_NODES.csv", "DIRECT_BOM_OCCURRENCES.csv",
            "DIRECT_LINK_CANDIDATES_V072.csv")
MAX_LEVEL = 10
MAX_ROWS = 100000

def read_csv(path):
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        return list(csv.DictReader(f))

def integer(value, default=0):
    try:
        return int(float(str(value).strip()))
    except (ValueError, TypeError):
        return default

def select_run():
    if not RUNS.is_dir():
        raise FileNotFoundError("CAD_RESULT/RUNS folder missing")
    candidates = [d for d in RUNS.iterdir() if d.is_dir()
                  and all((d / f).is_file() for f in REQUIRED)]
    if not candidates:
        raise FileNotFoundError("No V0.7.2 results containing all three required CSVs")
    return max(candidates, key=lambda d: d.name)

def normalize_links(links):
    """Consolidate duplicate evidence for same occurrence -> child drawing.
    Never invent any relation based on name similarity.
    """
    output = {}
    for x in links:
        occ = str(x.get("occurrence_id", "")).strip()
        child = integer(x.get("child_drawing_index"), -1)
        parent = integer(x.get("parent_drawing_index"), -1)
        if not occ or child <= 0 or parent <= 0 or parent == child:
            continue
        key = (occ, child)
        if key not in output:
            output[key] = dict(x)
        else:
            a = output[key]
            for field in ("match_rules", "source_fields"):
                existing = set(v.strip() for v in a.get(field, "").split("+") if v.strip())
                existing.update(v.strip() for v in x.get(field, "").split("+") if v.strip())
                a[field] = "+".join(sorted(existing))
    return list(output.values())

def hierarchy_rows(nodes, occurrences, links, drawing_titles=None):
    """Create a forest of drawing roots, then append their actual BOM items.
    Nodes without an incoming candidate are ROOT_CANDIDATE, never asserted root.
    Each child DWG is expanded only under a BOM occurrence that references it.
    """
    doc = {integer(d.get("drawing_index")): d for d in nodes}
    titles = {integer(t.get("drawing_index")):t
              for t in (drawing_titles or [])}
    def title_for(doc_id):
        record=titles.get(doc_id,{})
        part=str(record.get("drawing_part_number","") or "").strip()
        name=str(record.get("drawing_part_name","") or "").strip()
        # Avoid substituting the DWG filename as an asserted item identity.
        display=(part+"  "+name).strip()
        if not display:
            display="[도면 품명·품번 미검출 — 검토]"
        status=record.get("status","NOT_FOUND")
        return display,part,name,status
    parts = defaultdict(list)
    for i, occ in enumerate(occurrences):
        parent = integer(occ.get("parent_drawing_index"))
        if parent in doc:
            parts[parent].append((i, occ))
    for parent in parts:
        parts[parent].sort(key=lambda t:
            (integer(t[1].get("table_id", "").lstrip("T"), 1),
             integer(t[1].get("item_no"), 0),
             integer(t[1].get("subrow_id"), 1), t[0]))
    by_occ = defaultdict(list)
    incoming = set()
    for link in links:
        by_occ[str(link.get("occurrence_id", ""))].append(link)
        incoming.add(integer(link.get("child_drawing_index")))
    for key in by_occ:
        by_occ[key].sort(key=lambda e: (integer(e["child_drawing_index"]),
                                        str(e.get("status", ""))))
    root_candidates = sorted(set(doc) - incoming)
    if not root_candidates:
        root_candidates = sorted(doc)
    result = []
    expanded_roots = set()
    audit = []

    def append_entry(level, title, kind, parent, child, occ, link, warning=""):
        data = {
            "level": level, "node_title": title, "kind": kind,
            "part_number": occ.get("child_part_number", "") if occ else "",
            "drawing_number": occ.get("child_drawing_number", "") if occ else "",
            "part_name": occ.get("child_name", "") if occ else "",
            "qpa": occ.get("qpa", "") if occ else "",
            "parent_dwg": doc.get(parent, {}).get("drawing", "") if parent else "",
            "linked_dwg": doc.get(child, {}).get("drawing", "") if child else "",
            "link_status": link.get("status", "") if link else
                           ("NO_MATCHED_DWG" if kind == "PART" else warning),
            "match_rule": link.get("match_rules", "") if link else "",
            "occurrence_id": occ.get("occurrence_id", "") if occ else "",
            "table_id": occ.get("table_id", "") if occ else "",
            "item_no": occ.get("item_no", "") if occ else "",
            "warning": warning
        }
        if kind=="DRAWING":
            display,own_part,own_name,title_status=title_for(child)
            data["node_title"]=display
            data["part_number"]=own_part
            data["part_name"]=own_name
            data["warning"]=(warning+" | TITLE_"+title_status)
        result.append(data)
        return len(result)

    def visit_document(index, level, ancestors, origin=""):
        if len(result) >= MAX_ROWS:
            return
        expanded_roots.add(index)
        if index in ancestors:
            append_entry(level, "[CYCLE REVIEW] " + doc[index].get("drawing", ""),
                         "CYCLE", index, index, None, None, "CYCLE_REVIEW")
            return
        lineage = ancestors + (index,)
        for _, occ in parts.get(index, []):
            if len(result) >= MAX_ROWS:
                break
            oid = str(occ.get("occurrence_id", ""))
            child_edges = by_occ.get(oid, [])
            pname = str(occ.get("child_name", "") or "")
            pno = str(occ.get("child_part_number", "") or "")
            title = (pno + "  " + pname).strip() or ("부품 " + str(occ.get("item_no", "?")))
            if len(child_edges) == 1:
                edge = child_edges[0]
                ci = integer(edge.get("child_drawing_index"))
                status = str(edge.get("status", "REVIEW_ONLY"))
                append_entry(level, title, "PART", index, ci, occ, edge,
                             "연결 후보 — 현장 검증 필요")
                audit.append((oid, index, ci, status))
                if ci in doc:
                    visit_document(ci, level+1, lineage, origin=oid)
            elif len(child_edges) > 1:
                append_entry(level, title, "PART", index, None, occ, None,
                             "MULTIPLE_CHILD_DRAWINGS_REVIEW")
                for edge in child_edges:
                    ci = integer(edge.get("child_drawing_index"))
                    # Keep V0.7.3 multi-candidate expansion, but do not
                    # create a fake "[도면 후보]" level/row. Actual subparts
                    # occupy just the next column, with the same audit trail.
                    audit.append((oid, index, ci, edge.get("status", "")))
                    if ci in doc:
                        visit_document(ci, level+1, lineage, origin=oid)
            else:
                append_entry(level, title, "PART", index, None, occ, None)

    for root in root_candidates:
        if len(result) >= MAX_ROWS:
            break
        append_entry(0, "",
                     "DRAWING", None, root, None, None,
                     "ROOT_CANDIDATE_NOT_CONFIRMED")
        visit_document(root, 1, tuple())

    # A cycle or disconnected subgraph may have no zero-incoming root.
    for index in sorted(doc):
        if index not in expanded_roots and len(result) < MAX_ROWS:
            append_entry(0, "",
                         "DRAWING", None, index, None, None,
                         "UNREACHED_DRAWING_REVIEW")
            visit_document(index, 1, tuple())
    return result, audit, root_candidates

def xmltext(value):
    value = "" if value is None else str(value)
    return escape(value.replace("\x00", "").replace("\r", " "),
                  {'"': "&quot;"})

def col_letter(n):
    s = ""
    while n:
        n, remainder = divmod(n - 1, 26)
        s = chr(65 + remainder) + s
    return s

def make_xlsx(dest, tree, occ, links, stats):
    """Write an Excel workbook with ZIP/XML standard library; no dependencies."""
    levels = ["LEVEL_%02d" % i for i in range(MAX_LEVEL)]
    metadata = ["부품번호", "도면번호", "품명", "수량", "부모DWG",
                "연결DWG", "연결상태", "일치근거", "Occurrence_ID",
                "표ID", "표순번", "검토내용"]
    columns = levels + metadata
    rows1 = [columns]
    styles1 = [[1]*len(columns)]
    for r in tree:
        depth = min(max(integer(r["level"]), 0), MAX_LEVEL-1)
        indent = [""]*MAX_LEVEL
        indent[depth] = ("↳ " if depth else "") + r["node_title"]
        values = indent + [r["part_number"], r["drawing_number"],
                           r["part_name"], r["qpa"], r["parent_dwg"],
                           r["linked_dwg"], r["link_status"],
                           r["match_rule"], r["occurrence_id"],
                           r["table_id"], r["item_no"], r["warning"]]
        rows1.append(values)
        # Each level uses a stable distinct fill across all its rows.
        # First line of a drawing group is bold, with the SAME level fill.
        style = 2 + depth + (MAX_LEVEL if r.get("group_start") else 0)
        styles1.append([style]*len(columns))
    rows2 = [["검증 항목", "값"]] + [[key, val] for key,val in stats.items()]
    styles2 = [[1,1]] + [[0,0] for _ in stats]
    rows3 = [["Occurrence_ID", "상위DWG Index", "하위DWG Index", "상태"]]
    styles3 = [[1]*4]
    for record in links:
        rows3.append(list(record))
        styles3.append([0]*4)

    def sheet_xml(data, styles, widths):
        out=['<?xml version="1.0" encoding="UTF-8" standalone="yes"?>',
             '<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main">',
             '<sheetViews><sheetView showGridLines="0" workbookViewId="0">'
             '<pane ySplit="1" topLeftCell="A2" activePane="bottomLeft" state="frozen"/>'
             '</sheetView></sheetViews>']
        out.append('<cols>')
        for i,width in enumerate(widths,1):
            out.append('<col min="%d" max="%d" width="%s" customWidth="1"/>' %
                       (i,i,width))
        out.append('</cols><sheetData>')
        for ri, values in enumerate(data,1):
            out.append('<row r="%d" ht="%s">' % (ri,25 if ri==1 else 21))
            for ci,value in enumerate(values,1):
                if value is None or value == "":
                    continue
                ref=col_letter(ci)+str(ri)
                sty=styles[ri-1][ci-1]
                # Always use typed inline strings to avoid formula injection.
                out.append('<c r="%s" s="%d" t="inlineStr"><is><t>%s</t></is></c>' %
                           (ref,sty,xmltext(value)))
            out.append('</row>')
        out.append('</sheetData>')
        if data:
            out.append('<autoFilter ref="A1:%s%d"/>' %
                       (col_letter(len(data[0])),len(data)))
        out.append('</worksheet>')
        return "".join(out).encode("utf-8")

    # Excel styles: one light color per hierarchy level. Repeated levels
    # retain the same fill. Group-start bolding does not change the level fill.
    level_colors=[
        "FFE8F1FD",  # 0 pastel blue
        "FFE7F5EB",  # 1 pastel green
        "FFFFF0D9",  # 2 pastel amber
        "FFF0EAFE",  # 3 pastel lavender
        "FFE4F4F8",  # 4 pastel cyan
        "FFFCE8EE",  # 5 pastel rose
        "FFEDF4DC",  # 6 pastel lime
        "FFE6ECF4",  # 7 slate blue
        "FFF8EEE3",  # 8 sand
        "FFEAEFFD",  # 9 periwinkle
    ]
    fills=[
        '<fill><patternFill patternType="none"/></fill>',
        '<fill><patternFill patternType="gray125"/></fill>',
        '<fill><patternFill patternType="solid"><fgColor rgb="FF213E59"/></patternFill></fill>',
    ] + ['<fill><patternFill patternType="solid"><fgColor rgb="%s"/></patternFill></fill>' % c
         for c in level_colors[:MAX_LEVEL]]
    xfs=[
        '<xf numFmtId="0" fontId="0" fillId="0" borderId="0" xfId="0"/>',
        '<xf numFmtId="0" fontId="1" fillId="2" borderId="0" xfId="0" applyFill="1" applyFont="1"/>'
    ]
    for bold_group in (False,True):
        for depth in range(MAX_LEVEL):
            xfs.append('<xf numFmtId="0" fontId="%d" fillId="%d" borderId="0" xfId="0" applyFill="1" applyFont="%d"/>' %
                       (2 if bold_group else 0,depth+3,1 if bold_group else 0))
    styles_xml=('<?xml version="1.0" encoding="UTF-8" standalone="yes"?>'
        '<styleSheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main">'
        '<fonts count="3">'
        '<font><sz val="10"/><name val="맑은 고딕"/></font>'
        '<font><b/><sz val="10"/><color rgb="FFFFFFFF"/><name val="맑은 고딕"/></font>'
        '<font><b/><sz val="10"/><color rgb="FF19324D"/><name val="맑은 고딕"/></font>'
        '</fonts><fills count="%d">%s</fills>' % (len(fills),"".join(fills)) +
        '<borders count="1"><border><left/><right/><top/><bottom/><diagonal/></border></borders>'
        '<cellStyleXfs count="1"><xf numFmtId="0" fontId="0" fillId="0" borderId="0"/></cellStyleXfs>'
        '<cellXfs count="%d">%s</cellXfs></styleSheet>' % (len(xfs),"".join(xfs)))
    book_xml='''<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<workbook xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main"
xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships">
<sheets><sheet name="계층형_BOM" sheetId="1" r:id="rId1"/>
<sheet name="검증요약" sheetId="2" r:id="rId2"/>
<sheet name="연결근거" sheetId="3" r:id="rId3"/></sheets></workbook>'''
    rels='''<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
<Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet1.xml"/>
<Relationship Id="rId2" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet2.xml"/>
<Relationship Id="rId3" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet3.xml"/>
<Relationship Id="rId4" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>
</Relationships>'''
    rootrels='''<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
<Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="xl/workbook.xml"/>
</Relationships>'''
    types='''<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">
<Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>
<Default Extension="xml" ContentType="application/xml"/>
<Override PartName="/xl/workbook.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml"/>
<Override PartName="/xl/styles.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.styles+xml"/>
<Override PartName="/xl/worksheets/sheet1.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>
<Override PartName="/xl/worksheets/sheet2.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>
<Override PartName="/xl/worksheets/sheet3.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>
</Types>'''
    with zipfile.ZipFile(dest,"w",zipfile.ZIP_DEFLATED) as z:
        z.writestr("[Content_Types].xml",types)
        z.writestr("_rels/.rels",rootrels)
        z.writestr("xl/workbook.xml",book_xml)
        z.writestr("xl/_rels/workbook.xml.rels",rels)
        z.writestr("xl/styles.xml",styles_xml)
        z.writestr("xl/worksheets/sheet1.xml",
                   sheet_xml(rows1,styles1,[17]*MAX_LEVEL+[22,22,28,10,26,26,27,28,20,12,12,29]))
        z.writestr("xl/worksheets/sheet2.xml",
                   sheet_xml(rows2,styles2,[40,28]))
        z.writestr("xl/worksheets/sheet3.xml",
                   sheet_xml(rows3,styles3,[24,22,22,36]))

def main():
    if "--selftest" in sys.argv:
        nodes=[{"drawing_index":"1","drawing":"ASM-A.dwg"},
               {"drawing_index":"2","drawing":"ASM-B.dwg"}]
        occ=[{"occurrence_id":"O00001","parent_drawing_index":"1",
              "child_part_number":"ASM-B","child_name":"Assembly B","item_no":"1",
              "table_id":"T1","subrow_id":"1","qpa":"1"},
             {"occurrence_id":"O00002","parent_drawing_index":"2",
              "child_part_number":"PART-C","child_name":"Part C","item_no":"1",
              "table_id":"T1","subrow_id":"1","qpa":"2"}]
        link=[{"occurrence_id":"O00001","parent_drawing_index":"1",
               "child_drawing_index":"2","status":"FILENAME_EMBEDDED_REVIEW",
               "match_rules":"IDENTIFIER_INSIDE_FILENAME"}]
        fake_titles=[{"drawing_index":1,"drawing_part_number":"ASM-A",
                      "drawing_part_name":"ASSEMBLY_A","status":"FOUND"}]
        tree,audit,roots=hierarchy_rows(nodes,occ,link,fake_titles)
        assert tree[0]["node_title"]=="ASM-A  ASSEMBLY_A"
        assert tree[0]["kind"]=="DRAWING"
        assert roots==[1] and len([x for x in tree if x["kind"]=="PART"])==2
        assert any(x["level"]==2 and x["part_number"]=="PART-C" for x in tree)
        # One occurrence with 2 candidate DWGs: do not invent a middle
        # DRAWING row or extra hierarchy level.
        nodes2 = nodes + [{"drawing_index":"3","drawing":"ASM-D.dwg"}]
        occ2 = occ + [{"occurrence_id":"O00003","parent_drawing_index":"3",
                       "child_part_number":"PART-E","child_name":"Part E",
                       "item_no":"1","table_id":"T1","subrow_id":"1","qpa":"1"}]
        edges2 = link + [{"occurrence_id":"O00001",
                          "parent_drawing_index":"1",
                          "child_drawing_index":"3",
                          "status":"AMBIGUOUS_REVIEW",
                          "match_rules":"IDENTIFIER_INSIDE_FILENAME"}]
        tree2, audit2, roots2=hierarchy_rows(nodes2,occ2,edges2)
        assert not any("[도면 후보]" in x["node_title"] for x in tree2)
        assert not any(x["kind"]=="LINK_REVIEW" for x in tree2)
        assert any(x["part_number"]=="PART-C" and x["level"]==2 for x in tree2)
        assert any(x["part_number"]=="PART-E" and x["level"]==2 for x in tree2)
        assert len([x for x in audit2 if x[0]=="O00001"])==2
        print("V0.7.8|SELFTEST|PASS|BASELINE_AND_MULTIPLE_CANDIDATES")
        return 0
    run=select_run()
    nodes=read_csv(run/"DRAWING_NODES.csv")
    title_path=run/"DRAWING_TITLE_RESULTS.csv"
    drawing_titles=read_csv(title_path) if title_path.is_file() else []
    occ=read_csv(run/"DIRECT_BOM_OCCURRENCES.csv")
    links=normalize_links(read_csv(run/"DIRECT_LINK_CANDIDATES_V072.csv"))
    # Preserve strict links, even when absent from the extended V072 candidates.
    strict=run/"DIRECT_DRAWING_LINKS.csv"
    if strict.is_file():
        for x in read_csv(strict):
            if x.get("status")=="EXACT_CANDIDATE":
                x["match_rules"]=x.get("evidence","STRICT_EXACT")
                links=normalize_links(links+[x])
    tree,audit,roots=hierarchy_rows(nodes,occ,links,drawing_titles)
    stat={
        "버전":VERSION,"원본실행폴더":run.name,
        "DWG_수":len(nodes),"직속품목후보_수":len(occ),
        "도면대표품목_정상추출":sum(x.get("status")=="FOUND" for x in drawing_titles),
        "도면대표품목_미확정":len(nodes)-sum(x.get("status")=="FOUND" for x in drawing_titles),
        "직속연결후보_수":len(links),
        "연결포함_occurrence_수":len(set(x["occurrence_id"] for x in links)),
        "계층형_표시행_수":len(tree),"루트_후보_수":len(roots),
        "순환_경고_행수":sum(x["kind"]=="CYCLE" for x in tree),
        "확정_관계_건수":"0 (후보이며 현장 검증 필요)",
        "주의":"루트는 현재 입력 도면 집합 기준 후보. 확정 최상위 아님"
    }
    outfile=run/"BOM_HIERARCHY_V078.xlsx"
    make_xlsx(outfile,tree,occ,audit,stat)
    with (run/"BOM_HIERARCHY_V078.csv").open(
        "w",newline="",encoding="utf-8-sig") as f:
        keys=["level","node_title","kind","part_number","drawing_number",
              "part_name","qpa","parent_dwg","linked_dwg","link_status",
              "match_rule","occurrence_id","table_id","item_no","warning"]
        writer=csv.DictWriter(f,fieldnames=keys,extrasaction="ignore")
        writer.writeheader()
        writer.writerows(tree)
    (run/"HIERARCHY_SUMMARY_V078.json").write_text(
        json.dumps(stat,ensure_ascii=False,indent=2),encoding="utf-8")
    print("V0.7.8|DWG%d|직속품목%d|연결후보%d|계층행%d|엑셀OK" %
          (len(nodes),len(occ),len(links),len(tree)))
    print("OUTPUT="+str(outfile))
    return 0

if __name__=="__main__":
    try:
        sys.exit(main())
    except Exception as exc:
        print("V0.7.8|E99|"+type(exc).__name__+"|"+str(exc)[:90])
        sys.exit(99)
